apiVersion: apps/v1
kind: Deployment
metadata:
  name: demo-nginx
  namespace: test
  labels:
    app: demo-nginx
spec:
  # 初始副本数，可以修改这里实现扩缩容
  replicas: 2
  selector:
    matchLabels:
      app: demo-nginx
  # 滚动更新策略配置（重点）
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1        # 更新时最多额外新增几个Pod
      maxUnavailable: 0  # 更新过程中，最多允许多少旧Pod不可用，0代表更新时保证所有副本可用
  template:
    metadata:
      labels:
        app: demo-nginx
    spec:
      containers:
      - name: nginx
        image: nginx:1.30.5
        ports:
        - containerPort: 80
        resources:
          requests:
            cpu: 100m
            memory: 128Mi
          limits:
            cpu: 200m
            memory: 256Mi
---
# Service，NodePort类型，外部访问
apiVersion: v1
kind: Service
metadata:
  name: demo-nginx-svc
  namespace: test
spec:
  type: NodePort
  selector:
    app: demo-nginx
  ports:
  - port: 80
    targetPort: 80
    nodePort: 30081  # 端口范围 30000~32767
